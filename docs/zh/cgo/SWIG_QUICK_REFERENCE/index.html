    GoVector SWIG CGO 快速参考卡片¶

    快速开始¶

    1. 环境检查¶

    # 安装 SWIG
brew install swig  # macOS
sudo apt-get install swig  # Ubuntu

# 验证安装
swig -version
go version
gcc --version

    2. 构建 C API¶

    cd govector/capi

# 一键构建（Python 绑定）
./build.sh all

# 或单独构建
./build.sh python    # Python 绑定
./build.sh java      # Java 绑定
./build.sh csharp    # C# 绑定

# 清理
./build.sh clean

    3. 文件结构¶

    govector/capi/
├── govector_c.h          # C 头文件（API 定义）
├── govector_c.go         # Go 导出实现
├── govector.i            # SWIG 接口文件
├── build.sh              # 构建脚本
├── Makefile              # Make 构建配置
└── examples/             # 使用示例
    ├── simple_python_example.py
    ├── govector_client.py
    └── cpp_example.cpp

    API 参考¶

    核心数据类型¶

    距离度量¶

    typedef enum {
    DISTANCE_COSINE = 0,    // 余弦相似度
    DISTANCE_EUCLID = 1,    // 欧几里得距离
    DISTANCE_DOT = 2        // 点积
} DistanceType;

    错误码¶

    typedef enum {
    GOVECTOR_OK = 0,                    // 成功
    GOVECTOR_ERROR_GENERAL = 1,         // 一般错误
    GOVECTOR_ERROR_NOT_FOUND = 3,       // 未找到
    GOVECTOR_ERROR_STORAGE_FAILURE = 6, // 存储故障
    GOVECTOR_ERROR_MEMORY_ALLOC = 7     // 内存分配失败
} GovectorError;

    HNSW 参数¶

    typedef struct {
    int m;               // 最大连接数 (默认：16)
    int ef_construction; // 构建深度 (默认：200)
    int ef_search;       // 搜索深度 (默认：50)
    int k;               // 量化参数 (默认：2)
} HNSWParams;

    API 函数速查¶

    存储引擎¶

    函数 说明 返回值
    govector_storage_new(path, error) 创建存储引擎 StorageHandle
    govector_storage_close(handle) 关闭存储 GovectorError
    govector_storage_free(handle) 释放句柄 void

    集合管理¶

    函数 说明 返回值
    govector_collection_create(name, dim, metric, use_hnsw, params, storage, error) 创建集合 CollectionHandle
    govector_collection_load(name, storage, error) 加载集合 CollectionHandle
    govector_collection_drop(name, storage, error) 删除集合 GovectorError
    govector_collection_free(handle) 释放句柄 void
    govector_collection_count(handle) 获取点数 int

    点操作¶

    函数 说明 返回值
    govector_collection_upsert(handle, points, count, error) 插入/更新 GovectorError
    govector_collection_search(handle, query, dim, top_k, results, result_count, error) 搜索 GovectorError
    govector_collection_delete(handle, ids, count, deleted_count, error) 删除 GovectorError

    内存管理¶

    函数 说明
    govector_points_free(points, count) 释放点数组
    govector_scored_points_free(results, count) 释放搜索结果
    govector_error_free(error) 释放错误信息

    Python 示例¶

    基础用法¶

    import govector

# 初始化
error = govector.ErrorInfo()
storage = govector.govector_storage_new("test.db", error)

# 创建集合
params = govector.HNSWParams()
params.m = 16
collection = govector.govector_collection_create(
    "my_col", 3, govector.DISTANCE_COSINE,
    True, params, storage, error
)

# 获取数量
count = govector.govector_collection_count(collection)
print(f"Count: {count}")

# 清理
govector.govector_collection_free(collection)
govector.govector_storage_free(storage)

    封装类用法¶

    from govector_client import GovectorClient

client = GovectorClient("example.db")
client.connect()
client.create_collection("products", vector_dim=128)

# 搜索
query = [0.1] * 128
results = client.search("products", query, top_k=5)

client.disconnect()

    C++ 示例¶

    #include "govector_c.h"

// 创建存储
ErrorInfo error;
StorageHandle storage = govector_storage_new("test.db", &error);

// 创建集合
HNSWParams params = {16, 200, 50, 2};
CollectionHandle col = govector_collection_create(
    "my_col", 3, DISTANCE_COSINE,
    true, params, storage, &error
);

// 获取数量
int count = govector_collection_count(col);

// 清理
govector_collection_free(col);
govector_storage_free(storage);

    常见错误及解决方案¶

    错误 1: SWIG 未找到¶

    # 解决：安装 SWIG
brew install swig  # macOS
sudo apt-get install swig  # Ubuntu

    错误 2: 编译失败 - 找不到 Python 头文件¶

    # 解决：安装 Python 开发包
brew install python  # macOS
sudo apt-get install python3-dev  # Ubuntu

    错误 3: 段错误 (Segmentation Fault)¶

    # 调试：使用 gdb
gdb ./your_program
(gdb) run
(gdb) bt  # 查看堆栈

    错误 4: 内存泄漏¶

    # 检测：使用 Valgrind
valgrind --leak-check=full ./your_program

    性能优化技巧¶

    1. 批量操作¶

    # ✓ 好：批量插入
points = [...]  # 1000 个点
client.upsert("col", points)

# ✗ 差：单个插入
for point in points:
    client.upsert("col", [point])

    2. 零拷贝（高级）¶

    // 直接使用 C 内存，避免复制
func zeroCopySearch(cVector *C.float, dim int) {
    vector := (*[1 << 30]float32)(unsafe.Pointer(cVector))[:dim:dim]
    // 直接使用 vector...
}

    3. 连接池¶

    # 复用客户端实例
client = GovectorClient()  # 创建一次
for query in queries:
    results = client.search("col", query)  # 多次使用

    最佳实践清单¶

    ✅ 必须做¶

    • [ ] 总是检查返回值和错误
    • [ ] 及时释放所有资源（句柄、内存）
    • [ ] 使用 defer 确保清理代码执行
    • [ ] 在多线程环境中使用锁保护
    • [ ] 批量操作而非单次调用

    ❌ 避免做¶

    • [ ] 不要忽略错误返回值
    • [ ] 不要忘记释放 C 内存
    • [ ] 不要在多个线程中共享句柄（不加锁）
    • [ ] 不要传递 Go 指针给 C 长期保存
    • [ ] 不要频繁进行 CGO 调用
