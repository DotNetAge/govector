    GoVector SWIG CGO 接口集成指南¶

    本文档详细说明如何使用 SWIG (Simplified Wrapper and Interface Generator) 为 GoVector 向量数据库项目创建 CGO 兼容接口，以便其他语言（如 C/C++、Python、Java 等）能够调用 GoVector 的核心功能。

    概述¶

    为什么使用 SWIG？¶

    SWIG 是一个强大的工具，可以自动生成将 Go 代码与其他编程语言连接起来的胶水代码。对于 GoVector 项目，使用 SWIG 的优势包括：

    1. 多语言支持：一次编写，支持 Python、Java、C#、Ruby 等多种语言
    2. 类型安全：自动处理类型转换和内存管理
    3. 维护性好：接口定义清晰，易于维护和扩展
    4. 性能优化：生成的包装代码经过优化，性能损失最小

    架构设计¶

    ┌─────────────────────────────────────────────────────────┐
│                    调用方应用                            │

                     ▼

    环境准备¶

    系统要求¶

    • 操作系统：macOS, Linux, Windows (需要 MinGW)
    • Go 版本：1.25.1 或更高
    • SWIG 版本：4.0 或更高
    • C 编译器：gcc/clang (macOS 需要 Xcode Command Line Tools)

    安装 SWIG¶

    macOS¶

    # 使用 Homebrew
brew install swig

# 验证安装
swig -version

    Ubuntu/Debian¶

    sudo apt-get update
sudo apt-get install swig

    CentOS/RHEL¶

    sudo yum install swig

    Windows¶

    从 SWIG 官网 下载预编译的二进制文件，或使用 Chocolatey： 

    choco install swig

    安装 Python 开发包（可选，用于 Python 绑定）¶

    # macOS
brew install python

# Ubuntu/Debian
sudo apt-get install python3-dev

# 安装 Python C 类型文件
pip3 install cffi

    验证环境¶

    # 检查 Go 安装
go version

# 检查 SWIG 安装
swig -version

# 检查 C 编译器
gcc --version  # 或 clang --version

    项目结构改造¶

    在 govector 项目根目录下创建 capi (C API) 目录：

    govector/
├── api/
├── cmd/
├── core/
├── capi/                    # 新增：C API 层
│   ├── govector_c.h        # C 头文件（接口定义）
│   ├── govector_c.go       # Go 导出实现
│   ├── govector.i          # SWIG 接口文件
│   ├── govector_wrap.cxx   # SWIG 生成的包装文件（构建时生成）
│   └── govector_python.go  # Python 特定导出（可选）
├── docs/
└── go.mod

    创建 capi 目录¶

    cd govector
mkdir -p capi

    SWIG 接口文件编写¶

    1. 创建 SWIG 接口文件¶

    创建文件 govector/capi/govector.i：

    /* govector/capi/govector.i */
%module govector

/* 启用 Go 模块支持 */
%{
#include "govector_c.h"
%}

/* 引入 C 头文件 */
%include "govector_c.h"

/* Go 语言特定配置 */
%goheader("package capi")
%gopackage("capi")

/* 类型映射配置 */
%typemap(in) const char* {
    if ($input == NULL) {
        PyErr_SetString(PyExc_TypeError, "string expected");
        return NULL;
    }
    $1 = $input;
}

%typemap(out) const char* {
    $result = Py_BuildValue("s", $1);
}

/* 数组类型映射 */
%typemap(in) (float* vector, int dim) {
    if (!PyList_Check($input)) {
        PyErr_SetString(PyExc_TypeError, "list of floats expected");
        return NULL;
    }
    $2 = PyList_Size($input);
    $1 = (float*)malloc($2 * sizeof(float));
    for (int i = 0; i < $2; i++) {
        PyObject* item = PyList_GetItem($input, i);
        $1[i] = (float)PyFloat_AsDouble(item);
    }
}

%typemap(freememory) (float* vector, int dim) {
    free($1);
}

/* 错误处理 */
%exception {
    PyObject *result = $action;
    if (!result) {
        // Go 错误已设置到 Python 异常
        return NULL;
    }
    return result;
}

/* 要导出的 C 函数 */
%include "govector_c.h"

    C 头文件定义¶

    2. 创建 C 头文件¶

    创建文件 govector/capi/govector_c.h：

    /* govector/capi/govector_c.h */
#ifndef GOVECTOR_C_H
#define GOVECTOR_C_H

#include <stddef.h>
#include <stdint.h>
#include <stdbool.h>

#ifdef __cplusplus
extern "C" {
#endif

/* ============================================================================
 * 类型定义
 * ============================================================================ */

/* 距离度量类型 */
typedef enum {
    DISTANCE_COSINE = 0,    // 余弦相似度
    DISTANCE_EUCLID = 1,    // 欧几里得距离
    DISTANCE_DOT = 2        // 点积
} DistanceType;

/* HNSW 参数结构 */
typedef struct {
    int m;              // 每个节点的最大连接数
    int ef_construction; // 构建时的搜索深度
    int ef_search;      // 搜索时的深度
    int k;              // 量化参数
} HNSWParams;

/* 点数据结构 */
typedef struct {
    const char* id;           // 点 ID
    uint64_t version;         // 版本号
    float* vector;            // 向量数据
    int vector_dim;           // 向量维度
    const char* payload_json; // Payload JSON 字符串
} PointStruct;

/* 搜索结果结构 */
typedef struct {
    const char* id;           // 点 ID
    uint64_t version;         // 版本号
    float score;              // 相似度分数
    const char* payload_json; // Payload JSON 字符串
} ScoredPoint;

/* 集合句柄（不透明指针） */
typedef void* CollectionHandle;

/* 存储句柄（不透明指针） */
typedef void* StorageHandle;

/* ============================================================================
 * 错误码定义
 * ============================================================================ */
typedef enum {
    GOVECTOR_OK = 0,
    GOVECTOR_ERROR_GENERAL = 1,
    GOVECTOR_ERROR_INVALID_PARAM = 2,
    GOVECTOR_ERROR_NOT_FOUND = 3,
    GOVECTOR_ERROR_ALREADY_EXISTS = 4,
    GOVECTOR_ERROR_DIM_MISMATCH = 5,
    GOVECTOR_ERROR_STORAGE_FAILURE = 6,
    GOVECTOR_ERROR_MEMORY_ALLOC = 7
} GovectorError;

/* ============================================================================
 * 错误信息结构
 * ============================================================================ */
typedef struct {
    GovectorError code;
    const char* message;
} ErrorInfo;

/* ============================================================================
 * 存储引擎 API
 * ============================================================================ */

/**
 * @brief 创建新的存储引擎
 * @param db_path 数据库文件路径
 * @param error 输出错误信息
 * @return 存储句柄，失败返回 NULL
 */
StorageHandle govector_storage_new(const char* db_path, ErrorInfo* error);

/**
 * @brief 关闭存储引擎
 * @param handle 存储句柄
 * @return 错误码
 */
GovectorError govector_storage_close(StorageHandle handle);

/**
 * @brief 释放存储句柄
 * @param handle 存储句柄
 */
void govector_storage_free(StorageHandle handle);

/* ============================================================================
 * 集合管理 API
 * ============================================================================ */

/**
 * @brief 创建新的向量集合
 * @param name 集合名称
 * @param vector_dim 向量维度
 * @param metric 距离度量类型
 * @param use_hnsw 是否使用 HNSW 索引
 * @param hnsw_params HNSW 参数
 * @param storage 存储句柄
 * @param error 输出错误信息
 * @return 集合句柄，失败返回 NULL
 */
CollectionHandle govector_collection_create(
    const char* name,
    int vector_dim,
    DistanceType metric,
    bool use_hnsw,
    HNSWParams hnsw_params,
    StorageHandle storage,
    ErrorInfo* error
);

/**
 * @brief 加载已存在的集合
 * @param name 集合名称
 * @param storage 存储句柄
 * @param error 输出错误信息
 * @return 集合句柄，失败返回 NULL
 */
CollectionHandle govector_collection_load(
    const char* name,
    StorageHandle storage,
    ErrorInfo* error
);

/**
 * @brief 删除集合
 * @param name 集合名称
 * @param storage 存储句柄
 * @param error 输出错误信息
 * @return 错误码
 */
GovectorError govector_collection_drop(
    const char* name,
    StorageHandle storage,
    ErrorInfo* error
);

/**
 * @brief 释放集合句柄
 * @param handle 集合句柄
 */
void govector_collection_free(CollectionHandle handle);

/* ============================================================================
 * 点操作 API
 * ============================================================================ */

/**
 * @brief 插入或更新点
 * @param handle 集合句柄
 * @param points 点数组
 * @param count 点的数量
 * @param error 输出错误信息
 * @return 错误码
 */
GovectorError govector_collection_upsert(
    CollectionHandle handle,
    const PointStruct* points,
    int count,
    ErrorInfo* error
);

/**
 * @brief 向量搜索
 * @param handle 集合句柄
 * @param query_vector 查询向量
 * @param vector_dim 向量维度
 * @param top_k 返回结果数量
 * @param results 输出结果数组
 * @param result_count 输出结果数量
 * @param error 输出错误信息
 * @return 错误码
 */
GovectorError govector_collection_search(
    CollectionHandle handle,
    const float* query_vector,
    int vector_dim,
    int top_k,
    ScoredPoint** results,
    int* result_count,
    ErrorInfo* error
);

/**
 * @brief 删除点
 * @param handle 集合句柄
 * @param ids 点 ID 数组
 * @param count ID 数量
 * @param deleted_count 输出实际删除的数量
 * @param error 输出错误信息
 * @return 错误码
 */
GovectorError govector_collection_delete(
    CollectionHandle handle,
    const char** ids,
    int count,
    int* deleted_count,
    ErrorInfo* error
);

/**
 * @brief 获取集合中点的数量
 * @param handle 集合句柄
 * @return 点的数量
 */
int govector_collection_count(CollectionHandle handle);

/* ============================================================================
 * 内存管理 API
 * ============================================================================ */

/**
 * @brief 释放点数组
 * @param points 点数组
 * @param count 点的数量
 */
void govector_points_free(PointStruct* points, int count);

/**
 * @brief 释放搜索结果数组
 * @param results 结果数组
 * @param count 结果数量
 */
void govector_scored_points_free(ScoredPoint* results, int count);

/**
 * @brief 释放错误信息
 * @param error 错误信息结构
 */
void govector_error_free(ErrorInfo* error);

#ifdef __cplusplus
}
#endif

#endif /* GOVECTOR_C_H */
