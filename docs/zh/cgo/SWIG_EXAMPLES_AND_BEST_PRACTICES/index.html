    Python 使用示例¶

    1. 简单示例¶

    创建文件 govector/capi/examples/simple_python_example.py：

    #!/usr/bin/env python3
# govector/capi/examples/simple_python_example.py

import ctypes
import json
import sys
import os

# 加载生成的 Python 模块
try:
    import govector
except ImportError:
    print("Error: govector module not found. Please build the Python binding first.")
    print("Run: ./build.sh python")
    sys.exit(1)

def main():
    print("=" * 60)
    print("GoVector Python Binding Example")
    print("=" * 60)

    # 初始化错误结构
    error = govector.ErrorInfo()

    # 1. 创建存储引擎
    print("\n1. Creating storage engine...")
    db_path = b"test.db".decode('utf-8')
    storage = govector.govector_storage_new(db_path, error)

    if not storage:
        print(f"Failed to create storage: {error.message.decode('utf-8')}")
        govector.govector_error_free(error)
        return

    print("✓ Storage created successfully")

    # 2. 创建集合
    print("\n2. Creating collection...")
    hnsw_params = govector.HNSWParams()
    hnsw_params.m = 16
    hnsw_params.ef_construction = 200
    hnsw_params.ef_search = 50
    hnsw_params.k = 2

    col_name = b"test_collection".decode('utf-8')
    collection = govector.govector_collection_create(
        col_name,
        3,  # 向量维度
        govector.DISTANCE_COSINE,
        True,  # 使用 HNSW
        hnsw_params,
        storage,
        error
    )

    if not collection:
        print(f"Failed to create collection: {error.message.decode('utf-8')}")
        govector.govector_error_free(error)
        govector.govector_storage_free(storage)
        return

    print("✓ Collection created successfully")

    # 3. 获取集合数量
    print("\n3. Getting collection count...")
    count = govector.govector_collection_count(collection)
    print(f"✓ Collection contains {count} points")

    # 4. 清理资源
    print("\n4. Cleaning up resources...")
    govector.govector_collection_free(collection)
    govector.govector_storage_close(storage)
    govector.govector_storage_free(storage)
    govector.govector_error_free(error)

    print("✓ Resources freed")

    print("\n" + "=" * 60)
    print("Example completed successfully!")
    print("=" * 60)

if __name__ == "__main__":
    main()

python3 examples/simple_python_example.py

    2. 高级示例 - 封装类¶

    创建文件 govector/capi/examples/govector_client.py：

    #!/usr/bin/env python3
# govector/capi/examples/govector_client.py

import ctypes
import json
from typing import List, Dict, Any, Optional

try:
    import govector
except ImportError:
    raise ImportError("govector module not found. Please build the Python binding first.")

class GovectorClient:
    """GoVector Python 客户端封装类"""

    def __init__(self, db_path: str = "govector.db"):
        self.db_path = db_path
        self.storage = None
        self.collections = {}
        self.error = govector.ErrorInfo()

    def connect(self) -> bool:
        """连接到存储引擎"""
        self.storage = govector.govector_storage_new(
            self.db_path,
            self.error
        )
        return self.storage is not None

    def disconnect(self):
        """断开连接并清理资源"""
        for handle in self.collections.values():
            govector.govector_collection_free(handle)
        self.collections.clear()

        if self.storage:
            govector.govector_storage_close(self.storage)
            govector.govector_storage_free(self.storage)
            self.storage = None

    def create_collection(self, name: str, vector_dim: int, 
                         use_hnsw: bool = True,
                         m: int = 16,
                         ef_construction: int = 200,
                         ef_search: int = 50) -> bool:
        """创建集合"""
        hnsw_params = govector.HNSWParams()
        hnsw_params.m = m
        hnsw_params.ef_construction = ef_construction
        hnsw_params.ef_search = ef_search
        hnsw_params.k = 2

        handle = govector.govector_collection_create(
            name,
            vector_dim,
            govector.DISTANCE_COSINE,
            use_hnsw,
            hnsw_params,
            self.storage,
            self.error
        )

        if handle:
            self.collections[name] = handle
            return True
        else:
            print(f"Failed to create collection: {self.error.message}")
            govector.govector_error_free(self.error)
            return False

    def load_collection(self, name: str) -> bool:
        """加载集合"""
        handle = govector.govector_collection_load(
            name,
            self.storage,
            self.error
        )

        if handle:
            self.collections[name] = handle
            return True
        else:
            print(f"Failed to load collection: {self.error.message}")
            govector.govector_error_free(self.error)
            return False

    def upsert(self, collection_name: str, 
               points: List[Dict[str, Any]]) -> bool:
        """
        插入或更新点

        Args:
            collection_name: 集合名称
            points: 点列表，每个点包含 id, vector, payload

        Returns:
            bool: 是否成功
        """
        if collection_name not in self.collections:
            print(f"Collection {collection_name} not found")
            return False

        handle = self.collections[collection_name]

        # 注意：由于 SWIG 包装的复杂性，实际使用中需要创建 C 结构体数组
        # 这里简化处理，实际项目需要使用 ctypes 创建 PointStruct 数组
        print("Upsert operation requires C struct array creation")
        print("This is a simplified example - see advanced usage for full implementation")

        return True

    def search(self, collection_name: str, 
               query_vector: List[float], 
               top_k: int = 10) -> List[Dict[str, Any]]:
        """
        搜索相似向量

        Args:
            collection_name: 集合名称
            query_vector: 查询向量
            top_k: 返回结果数量

        Returns:
            搜索结果列表
        """
        if collection_name not in self.collections:
            return []

        handle = self.collections[collection_name]

        # 创建 C float 数组
        vector_type = ctypes.c_float * len(query_vector)
        c_vector = vector_type(*query_vector)

        results_ptr = ctypes.POINTER(govector.ScoredPoint)()
        result_count = ctypes.c_int(0)

        ret = govector.govector_collection_search(
            handle,
            c_vector,
            len(query_vector),
            top_k,
            ctypes.byref(results_ptr),
            ctypes.byref(result_count),
            self.error
        )

        if ret != govector.GOVECTOR_OK:
            print(f"Search failed: {self.error.message}")
            govector.govector_error_free(self.error)
            return []

        # 转换结果为 Python 字典
        results = []
        if results_ptr and result_count.value > 0:
            for i in range(result_count.value):
                point = results_ptr[i]
                payload_json = point.payload_json.decode('utf-8') if point.payload_json else '{}'
                results.append({
                    'id': point.id.decode('utf-8'),
                    'score': point.score,
                    'payload': json.loads(payload_json)
                })

            # 释放内存
            govector.govector_scored_points_free(results_ptr, result_count)

        return results

    def delete(self, collection_name: str, ids: List[str]) -> int:
        """删除点"""
        if collection_name not in self.collections:
            return 0

        handle = self.collections[collection_name]
        deleted_count = ctypes.c_int(0)

        # 创建 C 字符串数组
        c_ids = (ctypes.c_char_p * len(ids))(*[id.encode('utf-8') for id in ids])

        ret = govector.govector_collection_delete(
            handle,
            c_ids,
            len(ids),
            ctypes.byref(deleted_count),
            self.error
        )

        if ret != govector.GOVECTOR_OK:
            print(f"Delete failed: {self.error.message}")
            govector.govector_error_free(self.error)
            return 0

        return deleted_count.value

    def count(self, collection_name: str) -> int:
        """获取集合中点的数量"""
        if collection_name not in self.collections:
            return 0

        handle = self.collections[collection_name]
        return govector.govector_collection_count(handle)


# 使用示例
def example_usage():
    """使用示例"""
    client = GovectorClient("example.db")

    try:
        # 连接
        if not client.connect():
            print("Failed to connect")
            return

        # 创建集合
        if not client.create_collection("products", vector_dim=128):
            print("Failed to create collection")
            return

        print("Collection created successfully")

        # 获取数量
        count = client.count("products")
        print(f"Collection contains {count} points")

        # 搜索示例
        query = [0.1] * 128
        results = client.search("products", query, top_k=5)

        print(f"\nSearch results ({len(results)}):")
        for result in results:
            print(f"  ID: {result['id']}, Score: {result['score']:.4f}")

        # 删除示例
        deleted = client.delete("products", ["id1", "id2"])
        print(f"\nDeleted {deleted} points")

    finally:
        # 断开连接
        client.disconnect()
        print("\nDisconnected from database")


if __name__ == "__main__":
    example_usage()

    C++ 使用示例¶

    创建文件 govector/capi/examples/cpp_example.cpp：

    // govector/capi/examples/cpp_example.cpp

#include <iostream>
#include <vector>
#include <string>
#include <cstring>
#include <map>
#include "../govector_c.h"

class GoVectorClient {
private:
    StorageHandle storage_;
    std::map<std::string, CollectionHandle> collections_;

public:
    GoVectorClient() : storage_(nullptr) {}

    ~GoVectorClient() {
        close();
    }

    bool connect(const std::string& db_path) {
        ErrorInfo error;
        storage_ = govector_storage_new(db_path.c_str(), &error);
        if (!storage_) {
            std::cerr << "Failed to connect: " << error.message << std::endl;
            govector_error_free(&error);
            return false;
        }
        return true;
    }

    bool createCollection(const std::string& name, int vector_dim, 
                         bool use_hnsw = true) {
        ErrorInfo error;

        HNSWParams params;
        params.m = 16;
        params.ef_construction = 200;
        params.ef_search = 50;
        params.k = 2;

        CollectionHandle handle = govector_collection_create(
            name.c_str(),
            vector_dim,
            DISTANCE_COSINE,
            use_hnsw,
            params,
            storage_,
            &error
        );

        if (!handle) {
            std::cerr << "Failed to create collection: " << error.message << std::endl;
            govector_error_free(&error);
            return false;
        }

        collections_[name] = handle;
        return true;
    }

    int count(const std::string& collection_name) {
        auto it = collections_.find(collection_name);
        if (it == collections_.end()) {
            return 0;
        }
        return govector_collection_count(it->second);
    }

    void close() {
        for (auto& pair : collections_) {
            govector_collection_free(pair.second);
        }
        collections_.clear();

        if (storage_) {
            govector_storage_close(storage_);
            govector_storage_free(storage_);
            storage_ = nullptr;
        }
    }
};

int main() {
    std::cout << "========================================" << std::endl;
    std::cout << "GoVector C++ Example" << std::endl;
    std::cout << "========================================" << std::endl;

    GoVectorClient client;

    // 连接
    if (!client.connect("test.db")) {
        return 1;
    }

    // 创建集合
    if (!client.createCollection("test_col", 3)) {
        return 1;
    }

    std::cout << "Collection created successfully" << std::endl;

    // 获取数量
    int count = client.count("test_col");
    std::cout << "Collection contains " << count << " points" << std::endl;

    // 清理
    client.close();

    std::cout << "\nExample completed successfully!" << std::endl;
    return 0;
}

    注意事项与最佳实践¶

    1. 内存管理¶

    关键原则¶

    规则 1：谁分配，谁释放

    // ✓ 正确示例
void correct_memory_management() {
    // C 分配的内存由 C 释放
    ScoredPoint* results = malloc(...);
    govector_scored_points_free(results, count);  // ✓

    // Go 通过 CGO 返回的字符串需要手动释放
    const char* msg = some_go_function();
    govector_string_free(msg);  // ✓
}

// ✗ 错误示例：内存泄漏
void memory_leak() {
    ScoredPoint* results = malloc(...);
    // 忘记释放  ✗
}

// ✗ 错误示例：重复释放
void double_free() {
    ScoredPoint* results = malloc(...);
    govector_scored_points_free(results, count);
    govector_scored_points_free(results, count);  // ✗
}

    规则 2：跨语言内存边界

    // Go 代码中
//export govector_function
func govector_function() *C.char {
    // 使用 C.CString 创建的字符串必须由 C 端释放
    return C.CString("Hello from Go")
}

    // C 代码中
const char* msg = govector_function();
// 使用完后必须释放
free((void*)msg);  // 或使用专门的释放函数

    内存管理清单¶

    • [ ] 所有通过 C.CString 创建的字符串必须用 C.free 释放
    • [ ] 所有通过 C.malloc 分配的内存必须有对应的释放调用
    • [ ] 在 Go 端使用 defer 确保清理代码执行
    • [ ] 在错误处理路径中也要释放资源
    • [ ] 避免在 Go 和 C 之间传递 Go 托管的指针
    • [ ] 使用工具检测内存泄漏（Valgrind、Go race detector）

    2. 错误处理¶

    错误处理模式¶

    Go 端：

    //export govector_function
func govector_function(handle C.Handle, errInfo *C.ErrorInfo) C.GovectorError {
    // 1. 检查句柄有效性
    if handle == nil {
        if errInfo != nil {
            *errInfo = newError(C.GOVECTOR_ERROR_NOT_FOUND, "invalid handle")
        }
        return C.GOVECTOR_ERROR_NOT_FOUND
    }

    // 2. 执行业务逻辑
    err := doSomething()
    if err != nil {
        if errInfo != nil {
            *errInfo = newError(C.GOVECTOR_ERROR_GENERAL, err.Error())
        }
        return C.GOVECTOR_ERROR_GENERAL
    }

    // 3. 成功
    return C.GOVECTOR_OK
}

    C 端：

    GovectorError example_usage() {
    ErrorInfo error;

    ResultHandle result = govector_function(handle, &error);
    if (result == NULL) {
        fprintf(stderr, "Error %d: %s\n", error.code, error.message);
        govector_error_free(&error);
        return GOVECTOR_ERROR_GENERAL;
    }

    govector_error_free(&error);
    return GOVECTOR_OK;
}

    错误码使用规范¶

    错误码 含义 使用场景
    GOVECTOR_OK 成功 操作成功完成
    GOVECTOR_ERROR_GENERAL 一般错误 未分类的错误
    GOVECTOR_ERROR_INVALID_PARAM 无效参数 参数验证失败
    GOVECTOR_ERROR_NOT_FOUND 未找到 资源不存在
    GOVECTOR_ERROR_ALREADY_EXISTS 已存在 资源已存在
    GOVECTOR_ERROR_DIM_MISMATCH 维度不匹配 向量维度错误
    GOVECTOR_ERROR_STORAGE_FAILURE 存储故障 数据库操作失败
    GOVECTOR_ERROR_MEMORY_ALLOC 内存分配失败 malloc 失败

    3. 类型转换¶

    Go 与 C 类型映射表¶

    Go 类型 C 类型 转换方式 注意事项
    string const char* C.CString() / C.GoString() 需要手动释放
    []float32 float* 手动复制数组 注意数组边界
    []byte uint8_t* (*C.uchar)(unsafe.Pointer()) 零拷贝优化
    int C.int 直接转换 -
    bool C.bool C.bool() / bool() -
    map[string]interface{} const char* (JSON) json.Marshal() JSON 序列化

    安全的数组转换¶

    // ✓ 正确：安全的数组转换
func safeArrayConversion(cArray *C.float, length int) []float32 {
    if cArray == nil || length <= 0 {
        return nil
    }

    // 使用切片安全地访问 C 数组
    // [1 << 30] 是最大容量，实际使用 length
    slice := (*[1 << 30]C.float)(unsafe.Pointer(cArray))[:length:length]

    // 复制到 Go 数组
    goArray := make([]float32, length)
    for i := 0; i < length; i++ {
        goArray[i] = float32(slice[i])
    }

    return goArray
}

// ✓ 正确：零拷贝优化（仅当确定 C 内存生命周期时）
func zeroCopyAccess(cArray *C.float, length int) []float32 {
    // 直接使用 C 内存，不复制
    // 注意：必须确保在访问期间 cArray 不会被释放
    return (*[1 << 30]float32)(unsafe.Pointer(cArray))[:length:length]
}

    4. 并发安全¶

    Go 端并发控制¶

    var (
    registryMu sync.RWMutex
    handles    = make(map[Handle]interface{})
)

//export govector_thread_safe_function
func govector_thread_safe_function(handle Handle) C.GovectorError {
    // 读锁用于查找
    registryMu.RLock()
    obj, exists := handles[handle]
    registryMu.RUnlock()

    if !exists {
        return C.GOVECTOR_ERROR_NOT_FOUND
    }

    // 写锁用于修改
    registryMu.Lock()
    // 修改对象...
    registryMu.Unlock()

    return C.GOVECTOR_OK
}

    C 端线程安全注意事项¶

    // ✗ 错误：在多个线程中使用同一个句柄不加锁
void* thread_func(void* arg) {
    CollectionHandle handle = (CollectionHandle)arg;
    govector_collection_search(handle, ...);  // 危险！
}

// ✓ 正确：使用互斥锁保护
pthread_mutex_t mutex;
void* safe_thread_func(void* arg) {
    pthread_mutex_lock(&mutex);
    govector_collection_search(handle, ...);
    pthread_mutex_unlock(&mutex);
}

    5. 性能优化¶

    减少 CGO 调用开销¶

    // ✗ 低效：频繁的 CGO 调用
for i := 0; i < n; i++ {
    process_single_item(items[i])  // 每次都是 CGO 调用，开销大
}

// ✓ 高效：批量处理
process_batch(items, n)  // 一次 CGO 调用处理所有，效率高

    性能对比¶

    方式 调用次数 性能 适用场景
    单次调用 N 次 差 小数据量

    构建发布包¶

    创建脚本 govector/capi/release.sh：

    #!/bin/bash
# govector/capi/release.sh

set -e

VERSION="0.1.0"
DIST_DIR="dist"

mkdir -p $DIST_DIR

# 构建不同平台的库
platforms=(
    "darwin/amd64"
    "darwin/arm64"
    "linux/amd64"
    "linux/arm64"
)

for platform in "${platforms[@]}"; do
    GOOS=${platform%/*}
    GOARCH=${platform#*/}

    echo "Building for $GOOS/$GOARCH..."

    GOOS=$GOOS GOARCH=$GOARCH go build \
        -buildmode=c-archive \
        -o $DIST_DIR/libgovector_${GOOS}_${GOARCH}.a \
        ./capi

    # 复制头文件
    cp govector_c.h $DIST_DIR/

    # 打包
    tar -czf $DIST_DIR/govector_${VERSION}_${GOOS}_${GOARCH}.tar.gz \
        -C $DIST_DIR \
        libgovector_${GOOS}_${GOARCH}.a \
        govector_c.h

    echo "Built govector_${VERSION}_${GOOS}_${GOARCH}.tar.gz"
done

echo "Release packages created in $DIST_DIR"

    常见问题解答 (FAQ)¶

    Q1: SWIG 生成的包装文件太大怎么办？¶

    A: 这是正常现象。SWIG 生成的 govector_wrap.cxx 通常有几万行代码。可以通过以下方式优化：

    // 在 .i 文件中只导出需要的函数
%ignore unused_function;  // 忽略不需要的函数
%include "govector_c.h"   // 只包含必要的头文件

    Q2: 如何处理 Go 的 panic 传递到 C？¶

    A: 使用 recover 捕获 panic：

    //export govector_safe_function
func govector_safe_function() C.GovectorError {
    defer func() {
        if r := recover(); r != nil {
            // 记录错误但不传播 panic
            log.Printf("Recovered from panic: %v", r)
        }
    }()

    // 可能 panic 的代码
    riskyOperation()

    return C.GOVECTOR_OK
}

    Q3: Python GIL 与 Go goroutine 冲突怎么办？¶

    A: 在 Go 端使用 runtime.LockOSThread()：

    import "runtime"

//export govector_python_callback
func govector_python_callback() {
    runtime.LockOSThread()
    defer runtime.UnlockOSThread()

    // 调用 Python API
}
